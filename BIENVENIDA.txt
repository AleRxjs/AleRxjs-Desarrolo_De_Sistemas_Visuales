DOCKER COMPOSE:
    docker-compose up -d         Iniciar servicios
    docker-compose down          Detener servicios
    docker-compose ps            Ver estado
    docker-compose logs -f       Ver logs en tiempo real

MAKEFILE (si tienes make):
    make up                      Iniciar
    make down                    Detener
    make logs                    Ver logs
    make help                    Ver todos los comandos

SCRIPTS INTERACTIVOS:
    manage.bat                   Menu Windows
    ./manage.sh                  Menu Linux/Mac

💡 TIPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• Los cambios en código se reflejan automáticamente (hot reload)
• Ver logs: docker-compose logs -f backend
• Acceder a MongoDB: docker-compose exec mongodb mongosh
• Detener sin perder datos: docker-compose down
• Detener y limpiar todo: docker-compose down -v


❓ ¿PROBLEMAS?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Puerto ocupado:
    Cambiar en docker-compose.yml

Backend no responde:
    docker-compose restart backend

Frontend no carga:
    Verificar: curl http://localhost:5000/api/health

Más ayuda:
    Ver README.md → Sección "Troubleshooting"
